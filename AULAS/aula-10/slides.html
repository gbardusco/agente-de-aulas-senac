🔧 Funções com Parâmetros e Retorno

function gerarLeitura(min, max) {
    let valor = Math.random() * (max - min) + min;
    return Math.floor(valor);
}

let temp = gerarLeitura(15, 40);   // ex: 27
let umid = gerarLeitura(30, 90);   // ex: 65
console.log("Temp:", temp, "Umid:", umid);

Math.random() gera número entre 0 e 1. Multiplicamos pela faixa desejada e somamos o mínimo.

🔁 Loop for

for (let i = 0; i < 5; i++) {
    console.log("Sensor", i + 1);
}
// Saída: Sensor 1, Sensor 2, Sensor 3, Sensor 4, Sensor 5

🎯 querySelectorAll + Loop

// Seleciona TODOS os elementos com classe "sensor"
let sensores = document.querySelectorAll(".sensor");

for (let i = 0; i < sensores.length; i++) {
    let valor = gerarLeitura(15, 40);
    sensores[i].textContent = valor + "°C";

    if (valor > 30) {
        sensores[i].style.color = "red";
    } else {
        sensores[i].style.color = "green";
    }
}
